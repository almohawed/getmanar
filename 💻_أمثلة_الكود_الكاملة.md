## مثال 1: تقرير بسيط

### الكود:

```dart
import 'package:flutter/material.dart';
import 'package:masar_app/src/features/common/services/pdf_export_service.dart';
import 'package:masar_app/src/core/domain/models/user.dart';

class SimpleReportExample extends StatelessWidget {
  final List<User> students;
  final String className;

  const SimpleReportExample({
    required this.students,
    required this.className,
  });

  Future<void> _exportReport() async {
    // إنشاء خدمة PDF
    final pdfService = PdfExportService(
      schoolName: 'مدرسة النور الابتدائية',
      teacherName: 'أحمد محمد',
      principalName: 'خالد علي',
    );

    // تصدير التقرير
    await pdfService.printStudentNamesLog(
      students,
      className,
    );
  }

  @override
  Widget build(BuildContext context) {
    return ElevatedButton(
      onPressed: _exportReport,
      child: Text('تصدير كشف الأسماء'),
    );
  }
}
```

- ✅ كشف أسماء الطلاب

## مثال 2: تقرير متقدم

### الكود:

```dart
import 'package:flutter/material.dart';
import 'package:masar_app/src/features/common/services/pdf_export_service.dart';
import 'package:masar_app/src/core/domain/models/user.dart';
import 'package:masar_app/src/features/attendance/domain/student_attendance.dart';

class AdvancedReportExample extends StatelessWidget {
  final List<User> students;
  final String className;
  final List<StudentAttendance> attendanceRecords;

  const AdvancedReportExample({
    required this.students,
    required this.className,
    required this.attendanceRecords,
  });

  Future<void> _exportAttendanceReport() async {
    // إنشاء خدمة PDF مع خيارات متقدمة
    final pdfService = PdfExportService(
      schoolName: 'مدرسة النور الابتدائية',
      teacherName: 'أحمد محمد',
      principalName: 'خالد علي',
      signerTitle: 'معلم الفصل',           // تخصيص العنوان
      defaultShowClassInfo: true,          // إظهار معلومات الفصل
      managerOnlyFooter: false,            // إظهار كلا التوقيعين
    );

    // تصدير تقرير الحضور
    await pdfService.printDailyAttendanceLog(
      students,
      className,
      attendanceRecords: attendanceRecords,
    );
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        ElevatedButton(
          onPressed: _exportAttendanceReport,
          child: Text('تصدير كشف الحضور'),
        ),
        SizedBox(height: 10),
        Text(
          'عدد الطلاب: ${students.length}',
          style: TextStyle(fontSize: 14),
        ),
      ],
    );
  }
}
```

### النتيجة:
- ✅ كشف الحضور اليومي

- ✅ عنوان مخصص "معلم الفصل"

## مثال 3: تقرير مخصص كامل

### الكود:

```dart
import 'package:flutter/material.dart';
import 'package:masar_app/src/features/common/services/pdf_export_service.dart';
import 'package:cloud_firestore/cloud_firestore.dart';

class CustomReportExample extends StatefulWidget {
  final String schoolId;
  final String classId;

  const CustomReportExample({
    required this.schoolId,
    required this.classId,
  });

  @override
  _CustomReportExampleState createState() => _CustomReportExampleState();
}

class _CustomReportExampleState extends State<CustomReportExample> {
  bool _isLoading = false;
  String? _errorMessage;

  Future<void> _exportCustomReport() async {
    setState(() {
      _isLoading = true;
      _errorMessage = null;
    });

    try {
      // جلب البيانات من Firebase
      final studentsSnapshot = await FirebaseFirestore.instance
          .collection('Schools/${widget.schoolId}/Students')
          .where('classId', isEqualTo: widget.classId)
          .get();

      final students = studentsSnapshot.docs
          .map((doc) => User.fromFirestore(doc))
          .toList();

      // جلب معلومات المدرسة
      final schoolDoc = await FirebaseFirestore.instance
          .collection('Schools')
          .doc(widget.schoolId)
          .get();

      final schoolName = schoolDoc.data()?['name'] ?? 'مدرسة';

      // جلب معلومات الفصل
      final classDoc = await FirebaseFirestore.instance
          .collection('Schools/${widget.schoolId}/Classes')
          .doc(widget.classId)
          .get();

      final className = classDoc.data()?['name'] ?? 'فصل';

      // إنشاء خدمة PDF
      final pdfService = PdfExportService(
        schoolName: schoolName,
        teacherName: 'أحمد محمد',
        principalName: 'خالد علي',
      );

      // تحضير البيانات المخصصة
      final customData = students.asMap().entries.map((entry) {
        final index = entry.key + 1;
        final student = entry.value;
        return [
          index.toString(),
          student.name,
          student.identityNumber ?? '-',
          student.phoneNumber ?? '-',
          '', // ملاحظات
        ];
      }).toList();

      // تصدير التقرير المخصص
      await pdfService._printGenericReport(
        title: 'كشف بيانات الطلاب الكامل',
        columns: [
          'م',
          'اسم الطالب',
          'رقم الهوية',
          'رقم الجوال',
          'ملاحظات',
        ],
        data: customData,
        filenameSuffix: 'CustomStudentData',
        className: className,
        semester: 'الثاني 1447',
        grade: 'الأول الابتدائي',
        schoolName: schoolName,
        cellAlignments: {
          1: pw.Alignment.centerRight, // محاذاة الاسم لليمين
        },
        columnWidths: {
          0: 30,   // عرض عمود الرقم
          1: 150,  // عرض عمود الاسم
          2: 100,  // عرض عمود الهوية
          3: 100,  // عرض عمود الجوال
          4: 120,  // عرض عمود الملاحظات
        },
        landscape: false, // عمودي
      );

      // إظهار رسالة نجاح
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('✅ تم تصدير التقرير بنجاح'),
            backgroundColor: Colors.green,
          ),
        );
      }
    } catch (e) {
      setState(() {
        _errorMessage = 'خطأ في التصدير: $e';
      });

      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('❌ فشل التصدير: $e'),
            backgroundColor: Colors.red,
          ),
        );
      }
    } finally {
      setState(() {
        _isLoading = false;
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Card(
      child: Padding(
        padding: EdgeInsets.all(16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Text(
              'تصدير تقرير مخصص',
              style: TextStyle(
                fontSize: 18,
                fontWeight: FontWeight.bold,
              ),
            ),
            SizedBox(height: 16),
            if (_errorMessage != null)
              Container(
                padding: EdgeInsets.all(12),
                decoration: BoxDecoration(
                  color: Colors.red.shade50,
                  borderRadius: BorderRadius.circular(8),
                ),
                child: Text(
                  _errorMessage!,
                  style: TextStyle(color: Colors.red),
                ),
              ),
            SizedBox(height: 16),
            ElevatedButton.icon(
              onPressed: _isLoading ? null : _exportCustomReport,
              icon: _isLoading
                  ? SizedBox(
                      width: 20,
                      height: 20,
                      child: CircularProgressIndicator(
                        strokeWidth: 2,
                        valueColor: AlwaysStoppedAnimation(Colors.white),
                      ),
                    )
                  : Icon(Icons.picture_as_pdf),
              label: Text(
                _isLoading ? 'جاري التصدير...' : 'تصدير PDF',
              ),
              style: ElevatedButton.styleFrom(
                padding: EdgeInsets.symmetric(vertical: 16),
              ),
            ),
          ],
        ),
      ),
    );
  }
}
```

### النتيجة:
- ✅ تقرير مخصص كامل
- ✅ جلب البيانات من Firebase
- ✅ معالجة الأخطاء
- ✅ واجهة مستخدم تفاعلية
- ✅ رسائل نجاح/فشل

## معالجة الأخطاء

### مثال 1: معالجة أخطاء التحميل

```dart
Future<void> _exportReportWithErrorHandling() async {
  try {
    final pdfService = PdfExportService(
      schoolName: 'مدرسة النور',
      teacherName: 'أحمد محمد',
      principalName: 'خالد علي',
    );

    await pdfService.printStudentNamesLog(students, className);
    
    // نجح التصدير
    print('✅ تم التصدير بنجاح');
    
  } on FileSystemException catch (e) {
    // خطأ في الملفات
    print('❌ خطأ في الملفات: $e');
    _showError('فشل تحميل الملفات المطلوبة');
    
  } on FormatException catch (e) {
    // خطأ في التنسيق
    print('❌ خطأ في التنسيق: $e');
    _showError('خطأ في تنسيق البيانات');
    
  } catch (e) {
    // خطأ عام
    print('❌ خطأ غير متوقع: $e');
    _showError('حدث خطأ أثناء التصدير');
  }
}

void _showError(String message) {
  ScaffoldMessenger.of(context).showSnackBar(
    SnackBar(
      content: Text(message),
      backgroundColor: Colors.red,
      action: SnackBarAction(
        label: 'إعادة المحاولة',
        textColor: Colors.white,
        onPressed: _exportReportWithErrorHandling,
      ),
    ),
  );
}
```

## مثال كامل: شاشة تقارير

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
import 'package:masar_app/src/features/common/services/pdf_export_service.dart';

class ReportsScreen extends ConsumerStatefulWidget {
  final String schoolId;
  final String classId;

  const ReportsScreen({
    required this.schoolId,
    required this.classId,
  });

  @override
  _ReportsScreenState createState() => _ReportsScreenState();
}

class _ReportsScreenState extends ConsumerState<ReportsScreen> {
  bool _isExporting = false;

  Future<void> _exportReport(String reportType) async {
    setState(() => _isExporting = true);

    try {
      // جلب البيانات
      final students = await _fetchStudents();
      final schoolName = await _fetchSchoolName();
      final className = await _fetchClassName();

      // إنشاء خدمة PDF
      final pdfService = PdfExportService(
        schoolName: schoolName,
        teacherName: 'أحمد محمد',
        principalName: 'خالد علي',
      );

      // تصدير حسب النوع
      switch (reportType) {
        case 'names':
          await pdfService.printStudentNamesLog(students, className);
          break;
        case 'attendance':
          final records = await _fetchAttendanceRecords();
          await pdfService.printDailyAttendanceLog(
            students,
            className,
            attendanceRecords: records,
          );
          break;
        case 'behavior':
          final records = await _fetchBehaviorRecords();
          await pdfService.printBehaviorLog(
            students,
            className,
            records: records,
            filled: true,
          );
          break;
      }

      // نجح
      _showSuccess('تم تصدير التقرير بنجاح');
    } catch (e) {
      _showError('فشل التصدير: $e');
    } finally {
      setState(() => _isExporting = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text('التقارير'),
      ),
      body: ListView(
        padding: EdgeInsets.all(16),
        children: [
          _buildReportCard(
            title: 'كشف أسماء الطلاب',
            icon: Icons.people,
            onTap: () => _exportReport('names'),
          ),
          SizedBox(height: 12),
          _buildReportCard(
            title: 'كشف الحضور والغياب',
            icon: Icons.check_circle,
            onTap: () => _exportReport('attendance'),
          ),
          SizedBox(height: 12),
          _buildReportCard(
            title: 'كشف السلوك',
            icon: Icons.star,
            onTap: () => _exportReport('behavior'),
          ),
        ],
      ),
    );
  }

  Widget _buildReportCard({
    required String title,
    required IconData icon,
    required VoidCallback onTap,
  }) {
    return Card(
      child: ListTile(
        leading: Icon(icon, size: 32),
        title: Text(title),
        trailing: _isExporting
            ? SizedBox(
                width: 24,
                height: 24,
                child: CircularProgressIndicator(strokeWidth: 2),
              )
            : Icon(Icons.arrow_forward_ios),
        onTap: _isExporting ? null : onTap,
      ),
    );
  }

  void _showSuccess(String message) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(message),
        backgroundColor: Colors.green,
      ),
    );
  }

  void _showError(String message) {
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(message),
        backgroundColor: Colors.red,
      ),
    );
  }
```
