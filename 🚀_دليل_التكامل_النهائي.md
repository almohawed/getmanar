```dart
}) async {
```

```dart
// في smart_schedule_service.dart

Future<Map<String, List<ScheduleSlot>>> generateSchedule({
  required String schoolId,
  // ... parameters
}) async {
  // 1. جلب الحصص المحظورة من الحملة النشطة
  final campaignService = CampaignService();
  final activeCampaign = await campaignService.getActiveCampaign(schoolId);
  
  Map<String, List<BlockedSlot>> blockedSlots = {};
  if (activeCampaign != null) {
    blockedSlots = await campaignService.getBlockedSlots(
      schoolId: schoolId,
      campaignId: activeCampaign.id,
    );
  }

  // 2. توليد الجدول مع مراعاة الحصص المحظورة
  // ...

  // 3. تحليل النصاب
  final workloadService = WorkloadAnalysisService();
  final analysis = await workloadService.analyzeWorkload(schoolId);
  
  // 4. تطبيق التوصيات إذا كانت العدالة منخفضة
  if (analysis.fairnessScore < 70) {
    for (final rec in analysis.recommendations) {
      if (rec.autoApplicable) {
        await workloadService.applyRecommendation(
          schoolId: schoolId,
          recommendation: rec,
        );
      }
    }
  }

  // 5. تسجيل التعديل
  final modService = ModificationTrackingService();
  await modService.logModification(
    schoolId: schoolId,
    scheduleId: scheduleId,
    modifiedBy: userId,
```

## 🔐 الصلاحيات

### Firestore Security Rules

أضف هذه القواعد إلى `firestore.rules`:

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // الحملات التشاركية
    match /Schools/{schoolId}/ScheduleCampaigns/{campaignId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
                     get(/databases/$(database)/documents/Schools/$(schoolId)/Users/$(request.auth.uid)).data.role == 'educational_affairs_deputy';
      
      match /responses/{responseId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null;
      }
      
      match /blockedSlots/{slotId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null;
      }
    }
    
    // رؤى النصاب
    match /Schools/{schoolId}/WorkloadInsights/{insightId} {
      allow read: if request.auth != null;
      allow write: if request.auth != null && 
                     get(/databases/$(database)/documents/Schools/$(schoolId)/Users/$(request.auth.uid)).data.role in ['educational_affairs_deputy', 'principal'];
    }
    
    // سجل التعديلات
    match /Schools/{schoolId}/ScheduleModifications/{modificationId} {
      allow read: if request.auth != null;
      allow create: if request.auth != null;
      allow update, delete: if request.auth != null && 
                              get(/databases/$(database)/documents/Schools/$(schoolId)/Users/$(request.auth.uid)).data.role in ['educational_affairs_deputy', 'principal'];
    }
  }
}
```

## 📱 الإشعارات

### إعداد FCM

1. تأكد من إعداد Firebase Cloud Messaging
2. أضف الأذونات في `AndroidManifest.xml`:

```xml
<uses-permission android:name="android.permission.INTERNET"/>
<uses-permission android:name="android.permission.RECEIVE_BOOT_COMPLETED"/>
<uses-permission android:name="android.permission.VIBRATE" />
```

3. في `main.dart`:

```dart
import 'package:firebase_messaging/firebase_messaging.dart';

Future<void> _firebaseMessagingBackgroundHandler(RemoteMessage message) async {
  await Firebase.initializeApp();
  print('Handling background message: ${message.messageId}');
}

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await Firebase.initializeApp();
  
  FirebaseMessaging.onBackgroundMessage(_firebaseMessagingBackgroundHandler);
  
  runApp(MyApp());
}
```
