## 🚀 التكامل السريع

### 1. إضافة الأقسام إلى لوحة وكيل الشؤون

```dart
import 'package:flutter/material.dart';
import 'lib/src/features/schedule/presentation/create_campaign_screen.dart';
import 'lib/src/features/schedule/presentation/workload_analysis_screen.dart';
import 'lib/src/features/schedule/presentation/modifications_log_screen.dart';

class EducationalAffairsScreen extends StatelessWidget {
  final String schoolId;

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text('وكيل الشؤون التعليمية')),
      body: GridView.count(
        crossAxisCount: 2,
        padding: EdgeInsets.all(16),
        children: [
          _buildCard(
            context,
            '📋 الجدول التشاركي',
            Colors.blue,
            () => Navigator.push(context, MaterialPageRoute(
              builder: (_) => CreateCampaignScreen(schoolId: schoolId),
            )),
          ),
          _buildCard(
            context,
            '⚖️ توزيع النصاب',
            Colors.green,
            () => Navigator.push(context, MaterialPageRoute(
              builder: (_) => WorkloadAnalysisScreen(schoolId: schoolId),
            )),
          ),
          _buildCard(
            context,
            '📝 سجل التعديلات',
            Colors.orange,
            () => Navigator.push(context, MaterialPageRoute(
              builder: (_) => ModificationsLogScreen(schoolId: schoolId),
            )),
          ),
        ],
      ),
    );
  }

  Widget _buildCard(BuildContext context, String title, Color color, VoidCallback onTap) {
    return Card(
      child: InkWell(
        onTap: onTap,
        child: Container(
          decoration: BoxDecoration(
            gradient: LinearGradient(
              colors: [color.withOpacity(0.7), color],
            ),
          ),
          child: Center(
            child: Text(
              title,
              style: TextStyle(color: Colors.white, fontSize: 18, fontWeight: FontWeight.bold),
              textAlign: TextAlign.center,
            ),
          ),
        ),
      ),
    );
  }
}
```

