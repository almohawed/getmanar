def find_best_slot(schedule, teacher_schedule, cid, tid, sid, days, periods):
    """إيجاد أفضل فترة متاحة"""
    candidates = []
    for day in days:
        for period in periods:
            if schedule[cid][day][period] is None and not teacher_schedule[tid][day][period]:
                # حساب عدد نفس المادة في هذا اليوم
                same_subject = sum(1 for p in periods if schedule[cid][day][p] and schedule[cid][day][p][1] == sid)
                # تفضيل الأيام التي لا تحتوي على نفس المادة
                score = -same_subject
                candidates.append((day, period, score))
    
    if candidates:
        # ترتيب حسب الأفضلية
        candidates.sort(key=lambda x: x[2], reverse=True)
        return candidates[0][0], candidates[0][1]
    return None, None
