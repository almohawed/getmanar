```dart
class RemedialPlan {
  final String id;
  final String studentId;
  final String studentName;
  final String subjectId;
  final String subjectName;
  final String teacherId;
  final String teacherName;
  final String planType; // short, medium, long
  final String goals;
  final String actions;
  final DateTime startDate;
  final DateTime endDate;
  final String status; // active, in_progress, completed, overdue
  final double improvementScore; // 0-100
  final DateTime createdAt;
  final DateTime? updatedAt;

  RemedialPlan({
    required this.id,
    required this.studentId,
    required this.studentName,
    required this.subjectId,
    required this.subjectName,
    required this.teacherId,
    required this.teacherName,
    required this.planType,
    required this.goals,
    required this.actions,
    required this.startDate,
    required this.endDate,
    required this.status,
    this.improvementScore = 0,
    required this.createdAt,
    this.updatedAt,
  });

  factory RemedialPlan.fromFirestore(Map<String, dynamic> data, String id) {
    return RemedialPlan(
      id: id,
      studentId: data['studentId'] ?? '',
      studentName: data['studentName'] ?? '',
      subjectId: data['subjectId'] ?? '',
      subjectName: data['subjectName'] ?? '',
      teacherId: data['teacherId'] ?? '',
      teacherName: data['teacherName'] ?? '',
      planType: data['planType'] ?? 'short',
      goals: data['goals'] ?? '',
      actions: data['actions'] ?? '',
      startDate: (data['startDate'] as Timestamp).toDate(),
      endDate: (data['endDate'] as Timestamp).toDate(),
      status: data['status'] ?? 'active',
      improvementScore: (data['improvementScore'] ?? 0).toDouble(),
      createdAt: (data['createdAt'] as Timestamp).toDate(),
      updatedAt: data['updatedAt'] != null 
          ? (data['updatedAt'] as Timestamp).toDate() 
          : null,
    );
  }

  Map<String, dynamic> toFirestore() {
    return {
      'studentId': studentId,
      'studentName': studentName,
      'subjectId': subjectId,
      'subjectName': subjectName,
      'teacherId': teacherId,
      'teacherName': teacherName,
      'planType': planType,
      'goals': goals,
      'actions': actions,
      'startDate': Timestamp.fromDate(startDate),
      'endDate': Timestamp.fromDate(endDate),
      'status': status,
      'improvementScore': improvementScore,
      'createdAt': Timestamp.fromDate(createdAt),
      'updatedAt': updatedAt != null ? Timestamp.fromDate(updatedAt!) : null,
    };
  }
}
```

### 3. الخدمة (remedial_service.dart)

```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import '../domain/remedial_plan.dart';

class RemedialService {
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;

  // جلب الإحصائيات
  Future<RemedialStatistics> getStatistics(String schoolId) async {
    try {
      final plansSnapshot = await _firestore
          .collection('Schools/$schoolId/RemedialPlans')
          .get();

      final plans = plansSnapshot.docs
          .map((doc) => RemedialPlan.fromFirestore(doc.data(), doc.id))
          .toList();

      // حساب الخطط النشطة
      final activePlans = plans
          .where((p) => p.status == 'active' || p.status == 'in_progress')
          .length;

      // حساب الطلاب المستفيدين (unique)
      final beneficiaries = plans
          .map((p) => p.studentId)
          .toSet()
          .length;

      // حساب نسبة التحسن
      final improvementScores = plans
          .where((p) => p.improvementScore > 0)
          .map((p) => p.improvementScore);
      final improvementRate = improvementScores.isEmpty
          ? 0.0
          : improvementScores.reduce((a, b) => a + b) / improvementScores.length;

      // حساب الخطط المتأخرة
      final now = DateTime.now();
      final overduePlans = plans
          .where((p) => p.endDate.isBefore(now) && p.status != 'completed')
          .length;

      // حساب مؤشر الفعالية
      final completedPlans = plans.where((p) => p.status == 'completed').length;
      final totalPlans = plans.length;
      final effectivenessScore = totalPlans > 0
          ? ((completedPlans / totalPlans) * 50) + (improvementRate * 0.5)
          : 0.0;

      return RemedialStatistics(
        activePlans: activePlans,
        beneficiaries: beneficiaries,
        improvementRate: improvementRate,
        overduePlans: overduePlans,
        effectivenessScore: effectivenessScore,
      );
    } catch (e) {
      return RemedialStatistics(
        activePlans: 0,
        beneficiaries: 0,
        improvementRate: 0,
        overduePlans: 0,
        effectivenessScore: 0,
      );
    }
  }

  // توليد التوصيات الذكية
  List<String> generateRecommendations(RemedialStatistics stats) {
    final recommendations = <String>[];

    if (stats.overduePlans > 0) {
      recommendations.add(
        '⚠️ ${stats.overduePlans} خطط متأخرة تحتاج متابعة عاجلة',
      );
    }

    if (stats.improvementRate > 70) {
      recommendations.add('✅ نسبة التحسن ممتازة (${stats.improvementRate.toStringAsFixed(0)}%) - استمر!');
    } else if (stats.improvementRate < 50) {
      recommendations.add('⚠️ نسبة التحسن منخفضة (${stats.improvementRate.toStringAsFixed(0)}%) - راجع الخطط');
    }

    if (stats.effectivenessScore > 80) {
      recommendations.add('🎯 مؤشر الفعالية ممتاز - الخطط تعمل بشكل جيد');
    }

    if (recommendations.isEmpty) {
      recommendations.add('📊 لا توجد توصيات حالياً - استمر في المتابعة');
    }

    return recommendations;
  }

  // إنشاء خطة جديدة
  Future<void> createPlan(String schoolId, RemedialPlan plan) async {
    await _firestore
        .collection('Schools/$schoolId/RemedialPlans')
        .doc(plan.id)
        .set(plan.toFirestore());
  }

  // جلب جميع الخطط
  Stream<List<RemedialPlan>> getPlans(String schoolId) {
    return _firestore
        .collection('Schools/$schoolId/RemedialPlans')
        .orderBy('createdAt', descending: true)
        .snapshots()
        .map((snapshot) => snapshot.docs
            .map((doc) => RemedialPlan.fromFirestore(doc.data(), doc.id))
            .toList());
  }
}
```

```dart
Color getStatusColor(String status) {
  switch (status) {
    case 'active':
      return Colors.green;
    case 'in_progress':
      return Colors.blue;
    case 'completed':
      return Colors.teal;
    case 'overdue':
      return Colors.red;
    default:
      return Colors.grey;
  }
}

  switch (status) {
    case 'active':
      return Icons.play_circle;
    case 'in_progress':
      return Icons.pending;
    case 'completed':
      return Icons.check_circle;
    case 'overdue':
      return Icons.warning;
    default:
      return Icons.help;
  }
}
```
